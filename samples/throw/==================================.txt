// 1. Variable Declarations (Rubric Item 1.1)
int32 scalar i = 42;
int64 scalar long_val = 1000;
float32 scalar f = 3.14;
float16 scalar hf = 1.5;
bool scalar flag = true;
auto auto_val = 10;
string scalar name = "Quantel";

probe(i);
probe(long_val);
probe(f);
probe(flag);
probe(auto_val);
probe(name);

// 2. Constants (Rubric Item 1.5)
const int32 scalar MAX_VALUE = 100;
probe(MAX_VALUE);
// Semantic Check: Reassigning constant should fail (uncomment to test error)
// MAX_VALUE = 200; 

// 3. Pointers (Rubric Item 1.2)
int32 scalar target_var = 50;
int32 scalar *ptr = &target_var;
probe(ptr);

// 4. Records / Structures (Rubric Item 1.3)
record Point {
    int32 scalar x;
    int32 scalar y;
}

Point scalar p1;
p1.x = 10;
p1.y = 20;
probe(p1.x);
probe(p1.y);

// 5. Math Expressions (Rubric Item 5.a, 5.b)
// Simple Math
int32 scalar a = 10;
int32 scalar b = 20;
int32 scalar sum = a + b;
int32 scalar diff = b - a;
int32 scalar prod = a * b;
int32 scalar quot = b / a;
int32 scalar rem = 15 % 4;
int32 scalar pwr = 2 ^ 3;

probe(sum);
probe(diff);
probe(prod);
probe(quot);
probe(rem);
probe(pwr);

// Complex Math
float32 scalar complex_val = (10 + 5) * 2 / (3.0 ^ 1);
probe(complex_val);

// 6. Assignments & Augmented Assignments (Rubric Item 1.6)
int32 scalar count = 10;
count += 5;
probe(count);
count -= 2;
probe(count);
count *= 3;
probe(count);
count /= 4;
probe(count);

// 7. Boolean Expressions (Rubric Item 5.c, 5.d, 5.e)
// Simple Boolean
bool scalar b1 = 10 < 20;
bool scalar b2 = 10 > 20;
bool scalar b3 = 10 <= 10;
bool scalar b4 = 10 >= 5;
bool scalar b5 = 10 == 10;
bool scalar b6 = 10 != 5;
bool scalar b7 = !true;

probe(b1);
probe(b2);
probe(b3);
probe(b4);
probe(b5);
probe(b6);
probe(b7);

// Complex Logical
bool scalar complex_bool = (10 < 20) && (true || false) && !(5 == 6);
probe(complex_bool);

// 8. Control Flow (Rubric Item 1.7, 1.8)
// If-Else
if (count > 5) {
    probe("Count is greater than 5");
} else {
    probe("Count is 5 or less");
}

// While Loop (Event-controlled)
int32 scalar w = 0;
while (w < 3) {
    probe(w);
    w += 1;
}

// For Loop (Count-controlled)
for i in 0..3 {
    probe(i);
}

// Repeat-Until (Repeat-Until/Do-while)
int32 scalar r = 0;
repeat {
    probe(r);
    r += 1;
} until (r == 3);

// 9. Functions (Rubric Item 1.10)
// Declaration and Calling
func add(int32 scalar x, int32 scalar y) -> int32 scalar {
    return x + y;
}
probe(add(5, 7));

// Variable Scope
int32 scalar global_x = 100;
func scope_test() -> int32 scalar {
    int32 scalar local_x = 5;
    return global_x + local_x;
}
probe(scope_test());

// Function Calling Another Function
func square(int32 scalar n) -> int32 scalar {
    return n * n;
}
func sum_of_squares(int32 scalar a, int32 scalar b) -> int32 scalar {
    return square(a) + square(b);
}
probe(sum_of_squares(3, 4));

// Recursion
func factorial(int32 scalar n) -> int32 scalar {
    if (n <= 1) {
        return 1;
    }
    return n * factorial(n - 1);
}
probe(factorial(5));

// 10. Nested Statements (Rubric Item 1.11)
for i in 1..3 {
    if (i % 2 == 0) {
        probe("Even i");
        probe(i);
    } else {
        probe("Odd i");
        probe(i);
    }
}
